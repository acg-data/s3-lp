/* Scroll reveal */
const revealEls = document.querySelectorAll('.reveal');
const io = new IntersectionObserver((entries) => {
  entries.forEach((e, i) => {
    if (e.isIntersecting) {
      setTimeout(() => e.target.classList.add('in'), i * 55);
      io.unobserve(e.target);
    }
  });
}, { threshold: 0.08 });
revealEls.forEach(el => io.observe(el));

/* Sticky CTA */
const stickyCta = document.getElementById('sticky-cta');
const heroEl = document.querySelector('.hero');
new IntersectionObserver(([e]) => {
  stickyCta.classList.toggle('visible', !e.isIntersecting);
}, { threshold: 0.1 }).observe(heroEl);

/* Navigation */
function showBooking() {
  document.getElementById('landing-page').style.display = 'none';
  document.getElementById('booking-page').style.display = 'block';
  document.getElementById('success-page').style.display = 'none';
  window.scrollTo({ top: 0 });
}
function showLanding() {
  document.getElementById('landing-page').style.display = 'block';
  document.getElementById('booking-page').style.display = 'none';
  document.getElementById('success-page').style.display = 'none';
  window.scrollTo({ top: 0 });
}

/* Form submit */
function submitForm() {
  const required = ['f-first','f-last','f-email','f-phone'];
  let valid = true;
  required.forEach(id => {
    const el = document.getElementById(id);
    const empty = !el.value.trim();
    el.classList.toggle('error', empty);
    if (empty) valid = false;
    el.addEventListener('input', () => el.classList.remove('error'), { once: true });
  });
  const emailEl = document.getElementById('f-email');
  if (emailEl.value && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailEl.value.trim())) {
    emailEl.classList.add('error'); valid = false;
  }
  if (!valid) {
    document.querySelector('.field input.error')?.scrollIntoView({ behavior: 'smooth', block: 'center' });
    return;
  }
  document.getElementById('landing-page').style.display = 'none';
  document.getElementById('booking-page').style.display = 'none';
  document.getElementById('success-page').style.display = 'flex';
  window.scrollTo({ top: 0 });
}
</script>
